Purpose

This report compares the outputs of FADN2Footprint (v1.1.1) against the raw FADN data. An estimate is flagged in range if it falls within the literature min–max, within tolerance if the relative deviation from the literature midpoint is below a per-indicator threshold, or out of range otherwise.

    # Pivot longer to handle all herd variables at once
    tidyr::pivot_longer(

      names_to = "raw_variable",
      values_to = "value"
    ) |>
    # keep only variable with observed values
    dplyr::filter(value > 0) |>
    # Remove the suffix from the raw variable names
    dplyr::mutate(
      FADN_code_letter = sub("_AN$|_ALU$|_CN$|_ON$", "", raw_variable),
      pop_var = stringr::str_extract(raw_variable, "AN$|ALU$|CN$|ON$")
    ) |>
    # remove LTOT
    dplyr::filter(!grepl("LTOT", FADN_code_letter)) |>
    # pivot variables
    tidyr::pivot_wider(
      id_cols = c("ID", "COUNTRY", "YEAR", "SYS02", "FADN_code_letter"),
      names_from = pop_var,
      values_from = value
    ) |>
    # add livestock unit coefficients
    dplyr::left_join(
      data_extra$livestock |>
        dplyr::select(FADN_code_letter, livestock_unit_coef),
      by = "FADN_code_letter"
    ) |>
    # Calculate ALU where missing
    dplyr::mutate(
      ALU = dplyr::if_else(
        is.na(ALU) & !is.na(AN),
        AN * livestock_unit_coef,
        ALU
      )
    ) |>
    dplyr::mutate(
      temp_alu_calc = ALU / livestock_unit_coef
    ) |>
    # Calculate Qobs (Observed Quantity of animals in number of head)

        na.rm = TRUE
      ),
      Qobs_CN_ON = rowMeans(
        cbind(AN, temp_alu_calc, ON, CN),
        na.rm = TRUE
      ),
      raw_pop = Qobs_AN
    )
