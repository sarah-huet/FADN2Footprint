Comparison of sum of individual crop areas to total area variable

First, I want to check if I can recalculate the total utilised agricultural area variable (SE025) from the sum of crop areas with raw data.

# Extraction & Pivoting
# Filter, Pivot Long, and Split Code/Variable in one pass using tidyr::extract
raw_areas_crop <- raw_fadn |>
  dplyr::mutate(ID = as.character(ID), YEAR = as.character(YEAR)) |>
  dplyr::select(
    ID,
    COUNTRY,
    YEAR,
    SYS02,
    # keep only area (TA) variables
    #                  dplyr::matches("^C") & dplyr::matches("_TA$"),
    # keep only area (A) variables
    dplyr::matches("^C") & dplyr::matches("_A$|_TA$"),
    -dplyr::matches("^CTOT")
  ) |>
  # Pivot longer to handle all herd variables at once
  tidyr::pivot_longer(
    cols = -c("ID", "COUNTRY", "YEAR", "SYS02"),
    names_to = "raw_variable",
    values_to = "value"
  ) |>
  # keep only variable with observed values
  dplyr::filter(value > 0) |>
  # Remove the "_TA" suffix from the raw variable names
  dplyr::mutate(
    FADN_code_letter = sub("_TA$|_A$", "", raw_variable),
    area_var = stringr::str_extract(raw_variable, "TA$|A$")
  ) |>
  # pivot variables
  tidyr::pivot_wider(
    id_cols = c("ID", "COUNTRY", "YEAR", "SYS02", "FADN_code_letter"),
    names_from = area_var,
    values_from = value
  ) |>
  # estimate max area
  dplyr::mutate(
    area_ha = pmax(A, TA, na.rm = TRUE)
  )

raw_areas_crop_sum <- raw_areas_crop |>
  dplyr::filter(
    # remove mushroom, unused land, other land and rented out land
    !grepl("CMUSH|CUNUSEDLND|CLNDOTH|CRENTOUTLND", FADN_code_letter),
    # crops of which and sub-categories are removed
    !FADN_code_letter %in% crops_of_which,
    !FADN_code_letter %in% crops_sub_cat
  ) |>
  # remove aberrant bulgarian farm

  dplyr::summarise(
    crop_area_sum = sum(area_ha, na.rm = TRUE),

  )

Comparison of sum of individual crop areas between raw data and F2F output

Then, I want to estimate the share of arable and grassland removed from the package filter by comparing the sum of areas for each land use from the raw data and the F2F output.

raw_areas_crop_LU <- raw_areas_crop |>
  # add land use
  dplyr::left_join(
    data_extra$crops |>
      dplyr::select(FADN_code_letter, land_use_type),
    by = 'FADN_code_letter'
  )

raw_areas_crop_sum_LU <- raw_areas_crop_LU |>
  # sum
  dplyr::summarise(
    crop_area_sum_raw = sum(area_ha, na.rm = TRUE),
    .by = c(dplyr::all_of(id_cols), land_use_type)
  )
f2f_areas_crop_sum_LU <- my_object_GHGE@crop |>
  # sum
  dplyr::summarise(
    crop_area_sum_f2f = sum(area_ha, na.rm = TRUE),
    .by = c(dplyr::all_of(id_cols), land_use_type)
  )
# table of comparison
comp_crop_sum_raw_f2f <- raw_areas_crop_sum_LU |>
  dplyr::filter(crop_area_sum_raw > 0) |>
  dplyr::left_join(
    f2f_areas_crop_sum_LU,

  dplyr::mutate(
    rel_diff = (crop_area_sum_raw - dplyr::coalesce(crop_area_sum_f2f, 0)) /

  )
